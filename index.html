  <h1>Сравнение на Стилизиращи Продукти</h1>


  <div class="search-container">

    <input type="text" id="product-search" placeholder="Търси продукти..." oninput="searchProducts()" autocomplete="off">

    <div id="suggestions" class="search-suggestions"></div>

  </div>


  <button onclick="compareProducts()">Сравни</button>


  <div id="comparison"></div>


  <script>

    let products = [];

    const selectedProducts = [];


    async function fetchProductsFromSheet() {

      const sheetId = "1RCvnLdJRiiKOt-2d7YQK4RWl7VZultqwCVA_y3R0n4E";

      const sheetName = "Лист1";

      const url = `https://opensheet.elk.sh/${sheetId}/${encodeURIComponent(sheetName)}`;

      try {

        const response = await fetch(url);

        if (!response.ok) throw new Error("HTTP статус " + response.status);

        const data = await response.json();

        products = data.map(p => ({

          име: p.име,

          блясък: Number(p.блясък),

          фиксация: Number(p.фиксация),

          тип: p.тип,

          изображение: p.изображение,

          текстура: p.текстура,

          подходящ: p.подходящ,

          ползи: p.ползи

        }));

      } catch (error) {

        console.error("Грешка при зареждане на данни от Google Sheets:", error);

        alert("Неуспешно зареждане на продуктите. Моля, проверете дали таблицата е публична и правилно споделена.");

      }

    }


    fetchProductsFromSheet();


    function searchProducts() {

      const query = document.getElementById("product-search").value.toLowerCase();

      const suggestions = document.getElementById("suggestions");

      suggestions.innerHTML = "";


      if (query.length === 0) return;


      const matches = products.filter(p => p.име.toLowerCase().includes(query));

      matches.forEach(p => {

        const div = document.createElement("div");

        div.textContent = p.име;

        div.onclick = () => {

          if (!selectedProducts.includes(p)) {

            selectedProducts.push(p);

            document.getElementById("product-search").value = "";

            suggestions.innerHTML = "";

            renderSelected();

          }

        };

        suggestions.appendChild(div);

      });

    }


    function getBarColor(value) {

      if (value <= 4) return '#f44336'; // червено

      if (value <= 7) return '#ffeb3b'; // жълто

      return '#4caf50'; // зелено

    }


    function removeProduct(index) {

      selectedProducts.splice(index, 1);

      renderSelected();

    }


    function renderSelected() {

      const container = document.getElementById("comparison");

      if (selectedProducts.length === 0) {

        container.innerHTML = "<p>Няма избрани продукти.</p>";

        return;

      }


      let html = `<table><tr><th>Параметър</th>${selectedProducts.map((p, i) => 

        `<th style="position: relative;"><button class='remove-btn' onclick='removeProduct(${i})'>&times;</button><img src="${p.изображение}" class="product-image"><br>${p.име}</th>`).join('')}</tr>`;


      html += `<tr><td class="parameter">Тип</td>${selectedProducts.map(p => `<td>${p.тип}</td>`).join('')}</tr>`;


      const parameters = ["блясък", "фиксация"];

      parameters.forEach(param => {

        html += `<tr><td class="parameter">${param.charAt(0).toUpperCase() + param.slice(1)}</td>` +

          selectedProducts.map(p => {

            const value = p[param];

            const barWidth = value * 10;

            const barColor = getBarColor(value);

            return `<td><div class="rating-container"><div class="rating-bar" style="width: ${barWidth}%; background-color: ${barColor};"></div></div><span class="rating-label">${value}/10</span></td>`;

          }).join('') + '</tr>';

      });


      html += `<tr><td class="parameter">Текстура</td>${selectedProducts.map(p => `<td>${p.текстура}</td>`).join('')}</tr>`;

      html += `<tr><td class="parameter">Подходящ за</td>${selectedProducts.map(p => `<td>${p.подходящ}</td>`).join('')}</tr>`;

      html += `<tr><td class="parameter">Ползи</td>${selectedProducts.map(p => `<td>${p.ползи}</td>`).join('')}</tr>`;


      html += '</table>';

      container.innerHTML = html;

    }


    function compareProducts() {

      renderSelected();

    }
